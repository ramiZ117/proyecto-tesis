classDiagram
    direction TB

    class Usuario {
        rol
        activo
    }
    class Mesa {
        numero
        estado
    }
    class Comanda {
        estado
        notas
    }
    class DetalleComanda {
        cantidad
        precio
        estado
    }
    class Plato {
        nombre
        precioUsd
    }
    class Receta {
        cantidad
    }
    class Ingrediente {
        nombre
        stock
    }
    class Factura {
        numero
        total
    }

    Usuario "1" --> "0..*" Comanda : toma
    Mesa "1" --> "0..*" Comanda : atiende
    Comanda "1" *-- "1..*" DetalleComanda : compone
    DetalleComanda "*" --> "1" Plato : ordena
    Plato "1" *-- "0..*" Receta : define
    Receta "*" --> "1" Ingrediente : consume
    Comanda "1" --> "0..1" Factura : origina
    Usuario "1" --> "0..*" Factura : emite
